148static esp_err_t reg_write(uint8_t address, uint8_t value)
149{

153 .tx_buffer = tx,

165static esp_err_t reg_read(uint8_t address, uint8_t *value)
166{
167 uint8_t tx[2] = {(uint8_t)(address & 0x7F), 0x00};
168 uint8_t rx[2] = {0};
169 spi_transaction_t transaction = {
170 .length = sizeof(tx) * 8,
171 .tx_buffer = tx,
172 .rx_buffer = rx,
173 };
174 esp_err_t err = spi_device_transmit(s_spi, &transaction);
175 if (err == ESP_OK) {
176 *value = rx[1];
177 }
178 return err;
179}

188static esp_err_t fifo_write(const uint8_t *data, size_t length)
189{
190 if (length == 0 || length > LORA_MAX_PAYLOAD) {
191 return ESP_ERR_INVALID_SIZE;
192 }
193 uint8_t tx[LORA_MAX_PAYLOAD + 1];
194 tx[0] = REG_FIFO | 0x80;
195 memcpy(&tx[1], data, length);
196 spi_transaction_t transaction = {
197 .length = (length + 1) * 8,
198 .tx_buffer = tx,
199 };
200 return spi_device_transmit(s_spi, &transaction);
201}

308 s_mpu = i2c_bus_device_create(s_i2c_bus, MPU6050_ADDRESS, I2C_FREQUENCY_HZ);
309 if (s_mpu == NULL) {

312 }
313
314 uint8_t who_am_i = 0;
315 ESP_RETURN_ON_ERROR(i2c_bus_read_byte(s_mpu, 0x75, &who_am_i), TAG, "No se pudo leer WHO_AM_I del MPU6050");

325 ESP_RETURN_ON_ERROR(i2c_bus_write_byte(s_mpu, 0x1C, 0x00), TAG, "No se pudo configurar acelerómetro");
326 vTaskDelay(pdMS_TO_TICKS(100));
327

348 }
349 if (bme_status == BME68X_OK) {
350 s_bme_ready = true;
351 ESP_LOGI(TAG, "BME680 listo en 0x%02X", BME680_ADDRESS);
352 } else {
353 s_bme_ready = false;
354 ESP_LOGW(TAG, "BME680 no pudo configurarse (estado=%d); se enviaran solo datos del MPU6050", bme_status);
355 }
356 }
357
358 ESP_LOGI(TAG, "MPU6050 listo en 0x%02X; I2C SDA=%d SCL=%d",

361}

403static int32_t divide_round_signed(int32_t numerator, int32_t denominator)
404{
405 if (numerator >= 0) {
406 return (numerator + denominator / 2) / denominator;
407 }
408 return -((-numerator + denominator / 2) / denominator);

444static esp_err_t bme680_read(struct bme68x_data *sample)
445{
446 int8_t status = bme68x_set_op_mode(BME68X_FORCED_MODE, s_bme);
447 if (status != BME68X_OK) {
448 ESP_LOGE(TAG, "No se pudo iniciar medicion BME680: %d", status);

450 }
451

453 + ((uint32_t)s_bme_heater.heatr_dur * 1000U);
454 const uint32_t tick_us = (uint32_t)portTICK_PERIOD_MS * 1000U;
455 const TickType_t measurement_ticks = (measurement_us + tick_us - 1U) / tick_us;
456 vTaskDelay(measurement_ticks);
457
458 uint8_t fields = 0;
459 status = bme68x_get_data(BME68X_FORCED_MODE, sample, &fields, s_bme);
460 if (status != BME68X_OK || fields == 0) {
461 ESP_LOGE(TAG, "No se obtuvo una lectura BME680 valida (estado=%d, campos=%u)", status, fields);
462 return ESP_FAIL;
463 }
464 return ESP_OK;
465}

479static esp_err_t build_payload(uint32_t sequence, char *payload, size_t capacity, size_t *length)
480{
481 mpu6050_raw_t mpu = {0};
482 struct bme68x_data bme = {0};
483 ESP_RETURN_ON_ERROR(mpu6050_read(&mpu), TAG, "No se pudo leer MPU6050");
484
485 bool include_bme = s_bme_ready;
486 if (include_bme && bme680_read(&bme) != ESP_OK) {
487 ESP_LOGW(TAG, "Se perdio comunicacion con BME680; se continuara solo con MPU6050");
488 s_bme_ready = false;
489 include_bme = false;
490 }
491
492 const int32_t ax_hundredths = divide_round_signed((int32_t)mpu.ax * 100, 16384);
493 const int32_t ay_hundredths = divide_round_signed((int32_t)mpu.ay * 100, 16384);
494 const int32_t az_hundredths = divide_round_signed((int32_t)mpu.az * 100, 16384);
495 const int32_t gx_tenths = divide_round_signed((int32_t)mpu.gx * 10, 131);
496 const int32_t gy_tenths = divide_round_signed((int32_t)mpu.gy * 10, 131);
497 const int32_t gz_tenths = divide_round_signed((int32_t)mpu.gz * 10, 131);
498 const float mpu_temperature_c = ((float)mpu.temperature / 340.0f) + 36.53f;
499 const int32_t mpu_temp_tenths = float_to_tenths(mpu_temperature_c);
500 int written;
501
502 if (include_bme) {
503 const int32_t temp_tenths = float_to_tenths(bme.temperature);
504 const int32_t pressure_tenths_hpa = float_to_tenths(bme.pressure / 100.0f);
505 const int32_t humidity_tenths = float_to_tenths(bme.humidity);
506 const uint32_t gas_ohms = (uint32_t)(bme.gas_resistance + 0.5f);
507 written = snprintf(payload, capacity,
508 "SEQ:%lu TEMP:%s%d.%d PRES:%s%d.%d HUM:%s%d.%d GAS:%lu AX:%s%d.%02d AY:%s%d.%02d AZ:%s%d.%02d GX:%s%d.%d GY:%s%d.%d GZ:%s%d.%d TM:%s%d.%d",
509 (unsigned long)sequence,
510 temp_tenths < 0 ? "-" : "", magnitude(temp_tenths) / 10, magnitude(temp_tenths) % 10,
511 pressure_tenths_hpa < 0 ? "-" : "", magnitude(pressure_tenths_hpa) / 10, magnitude(pressure_tenths_hpa) % 10,
512 humidity_tenths < 0 ? "-" : "", magnitude(humidity_tenths) / 10, magnitude(humidity_tenths) % 10,
513 (unsigned long)gas_ohms,
514 ax_hundredths < 0 ? "-" : "", magnitude(ax_hundredths) / 100, magnitude(ax_hundredths) % 100,
515 ay_hundredths < 0 ? "-" : "", magnitude(ay_hundredths) / 100, magnitude(ay_hundredths) % 100,
516 az_hundredths < 0 ? "-" : "", magnitude(az_hundredths) / 100, magnitude(az_hundredths) % 100,
517 gx_tenths < 0 ? "-" : "", magnitude(gx_tenths) / 10, magnitude(gx_tenths) % 10,
518 gy_tenths < 0 ? "-" : "", magnitude(gy_tenths) / 10, magnitude(gy_tenths) % 10,
519 gz_tenths < 0 ? "-" : "", magnitude(gz_tenths) / 10, magnitude(gz_tenths) % 10,
520 mpu_temp_tenths < 0 ? "-" : "", magnitude(mpu_temp_tenths) / 10, magnitude(mpu_temp_tenths) % 10);
521 } else {
522 written = snprintf(payload, capacity,
523 "SEQ:%lu AX:%s%d.%02d AY:%s%d.%02d AZ:%s%d.%02d GX:%s%d.%d GY:%s%d.%d GZ:%s%d.%d TM:%s%d.%d",
524 (unsigned long)sequence,
525 ax_hundredths < 0 ? "-" : "", magnitude(ax_hundredths) / 100, magnitude(ax_hundredths) % 100,
526 ay_hundredths < 0 ? "-" : "", magnitude(ay_hundredths) / 100, magnitude(ay_hundredths) % 100,
527 az_hundredths < 0 ? "-" : "", magnitude(az_hundredths) / 100, magnitude(az_hundredths) % 100,
528 gx_tenths < 0 ? "-" : "", magnitude(gx_tenths) / 10, magnitude(gx_tenths) % 10,
529 gy_tenths < 0 ? "-" : "", magnitude(gy_tenths) / 10, magnitude(gy_tenths) % 10,
530 gz_tenths < 0 ? "-" : "", magnitude(gz_tenths) / 10, magnitude(gz_tenths) % 10,
531 mpu_temp_tenths < 0 ? "-" : "", magnitude(mpu_temp_tenths) / 10, magnitude(mpu_temp_tenths) % 10);
532 }
533
534 if (written < 0 || (size_t)written >= capacity || written > LORA_MAX_PAYLOAD) {
535 ESP_LOGE(TAG, "Payload demasiado largo (%d bytes; limite %d)", written, LORA_MAX_PAYLOAD);
536 return ESP_ERR_INVALID_SIZE;
537 }
538 *length = (size_t)written;
539 return ESP_OK;
540}

551static esp_err_t send_packet(const uint8_t *data, size_t length)
552{
553 if (length == 0 || length > LORA_MAX_PAYLOAD) {
554 return ESP_ERR_INVALID_SIZE;
555 }
556 ESP_RETURN_ON_ERROR(radio_set_mode(MODE_STDBY), TAG, "No se pudo entrar a standby");
557 ESP_RETURN_ON_ERROR(reg_write(REG_IRQ_FLAGS, 0xFF), TAG, "No se pudieron limpiar IRQ");
558 ESP_RETURN_ON_ERROR(reg_write(REG_FIFO_ADDR_PTR, 0x00), TAG, "No se pudo posicionar FIFO");
559 ESP_RETURN_ON_ERROR(fifo_write(data, length), TAG, "No se pudo escribir el paquete");
560 ESP_RETURN_ON_ERROR(reg_write(REG_PAYLOAD_LENGTH, (uint8_t)length), TAG, "No se pudo establecer longitud");
561 ESP_RETURN_ON_ERROR(radio_set_mode(MODE_TX), TAG, "No se pudo iniciar TX");
562
563 for (int i = 0; i < 500; ++i) {
564 uint8_t irq = 0;
565 ESP_RETURN_ON_ERROR(reg_read(REG_IRQ_FLAGS, &irq), TAG, "No se pudo leer IRQ");
566 if (irq & IRQ_TX_DONE) {
567 ESP_RETURN_ON_ERROR(reg_write(REG_IRQ_FLAGS, IRQ_TX_DONE), TAG, "No se pudo limpiar TX_DONE");
568 ESP_RETURN_ON_ERROR(radio_set_mode(MODE_STDBY), TAG, "No se pudo regresar a standby");
569 return ESP_OK;
570 }
571 vTaskDelay(pdMS_TO_TICKS(10));
572 }
573
574 (void)radio_set_mode(MODE_STDBY);
575 ESP_LOGE(TAG, "Tiempo agotado esperando TX_DONE");
576 return ESP_ERR_TIMEOUT;
577}

588void app_main(void)
589{
590 gpio_config_t led_config = {
591 .pin_bit_mask = 1ULL << LED_PIN,
592 .mode = GPIO_MODE_OUTPUT,
593 };
594 ESP_ERROR_CHECK(gpio_config(&led_config));
595 gpio_set_level(LED_PIN, 0);
596
597 ESP_ERROR_CHECK(i2c_sensors_init());
598 ESP_ERROR_CHECK(radio_init());
599
600 uint32_t sequence = 0;
601 while (true) {
602 char payload[LORA_MAX_PAYLOAD + 1];
603 size_t payload_length = 0;
604 esp_err_t err = build_payload(sequence, payload, sizeof(payload), &payload_length);
605 if (err == ESP_OK) {
606 gpio_set_level(LED_PIN, 1);
607 err = send_packet((const uint8_t *)payload, payload_length);
608 gpio_set_level(LED_PIN, 0);
609 if (err == ESP_OK) {
610 ESP_LOGI(TAG, "TX #%lu (%u bytes): %s", (unsigned long)sequence,
611 (unsigned)payload_length, payload);
612 ++sequence;
613 } else {
614 ESP_LOGE(TAG, "Fallo la transmision: %s", esp_err_to_name(err));
